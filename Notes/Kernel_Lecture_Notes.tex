\part{Foundations: Hilbert Spaces and Inner Products}

\section{Motivation and Overview}

Kernels are fundamental tools in machine learning that allow us to:
\begin{itemize}
    \item Compare objects based on their features without explicitly computing feature representations
    \item Control model smoothness and generalization to avoid overfitting
    \item Extend linear methods to non-linear settings through feature space transformations
\end{itemize}

The key insight is that many algorithms depend only on inner products between data points, which we can compute implicitly through a kernel function.

\section{Inner Products and Norms}

\subsection{Inner Product Spaces}

\begin{definition}[Inner Product]
Let $H$ be a vector space over $\mathbb{R}$. A function $\langle \cdot, \cdot \rangle_H : H \times H \to \mathbb{R}$ is an inner product if it satisfies:

\begin{enumerate}
    \item \textbf{Linearity in the first argument:} 
    \[
    \langle \alpha f + \beta g, h \rangle_H = \alpha \langle f, h \rangle_H + \beta \langle g, h \rangle_H
    \]
    for all $\alpha, \beta \in \mathbb{R}$ and $f, g, h \in H$.
    
    \item \textbf{Symmetry:} $\langle f, g \rangle_H = \langle g, f \rangle_H$ for all $f, g \in H$.
    
    \item \textbf{Positive Definiteness:} $\langle f, f \rangle_H \geq 0$ for all $f \in H$, with equality if and only if $f = 0$.
\end{enumerate}
\end{definition}

The inner product induces a norm on $H$:
\[
\|f\|_H := \sqrt{\langle f, f \rangle_H}
\]

\subsection{Norms on Vector Spaces}

\begin{definition}[Norm]
Let $F$ be a vector space over $\mathbb{R}$. A function $\|\cdot\|_F : F \to [0, \infty)$ is a norm if it satisfies:

\begin{enumerate}
    \item \textbf{Separation of points:} $\|f\|_F = 0 \Leftrightarrow f = 0$
    \item \textbf{Positive homogeneity:} $\|\lambda f\|_F = |\lambda| \|f\|_F$ for all $\lambda \in \mathbb{R}$ and $f \in F$
    \item \textbf{Triangle inequality:} $\|f + g\|_F \leq \|f\|_F + \|g\|_F$ for all $f, g \in F$
\end{enumerate}
\end{definition}

Every normed vector space carries an induced metric: $d(f, g) = \|f - g\|_F$. Conversely, every inner product space is a normed vector space with the norm induced by the inner product.

\subsection{Inner Products on \texorpdfstring{$\mathbb{R}^n$}{Rn}}

\begin{definition}[Inner Product on $\mathbb{R}^n$]
A symmetric positive definite matrix $M \in \mathbb{R}^{n \times n}$ defines an inner product on $\mathbb{R}^n$:
\[
\langle x, y \rangle_M := x^T M y \quad \text{for all } x, y \in \mathbb{R}^n
\]

The special case $M = I$ gives the standard Euclidean inner product: $\langle x, y \rangle = x^T y$.
\end{definition}

\section{Hilbert Spaces and Completeness}

\subsection{Cauchy Sequences and Completeness}

A sequence $\{f_n\}_{n=1}^{\infty}$ is Cauchy with respect to the induced metric if for every $\varepsilon > 0$, there exists $N$ such that for all $n, m > N$:
\[
\|f_n - f_m\|_F < \varepsilon
\]

Every convergent sequence is Cauchy, but not all Cauchy sequences converge. For example, the sequence $1, 1.4, 1.41, 1.414, \ldots$ is Cauchy in $\mathbb{Q}$ but converges (to $\sqrt{2}$) only in $\mathbb{R}$.

\begin{definition}[Complete Metric Space]
A normed vector space $F$ is complete if every Cauchy sequence $\{f_n\}$ in $F$ converges to some $f \in F$ (i.e., $\lim_{n \to \infty} \|f_n - f\|_F = 0$).
\end{definition}

\begin{definition}[Hilbert Space]
A Hilbert space is a complete inner product space. That is, it is a vector space with an inner product where every Cauchy sequence converges.

More formally, a Banach space is a complete normed vector space, while a Hilbert space is a complete inner product space.
\end{definition}

\subsection{Orthogonality and Orthonormal Basis}

\begin{definition}[Orthogonal Complement]
For a set $M \subseteq H$ in an inner product space, its orthogonal complement is:
\[
M^{\perp} := \{g \in H : \langle f, g \rangle = 0 \text{ for all } f \in M\}
\]

The orthogonal complement is always a closed linear subspace. If $M$ is a linear subspace, then $M \cap M^{\perp} = \{0\}$ (discrete intersection).
\end{definition}

\begin{definition}[Orthonormal Basis]
An orthonormal basis of a Hilbert space $H$ is a family $\{e_k\}_{k \in B}$ of elements satisfying:

\begin{enumerate}
    \item \textbf{Orthogonality:} $\langle e_i, e_j \rangle = 0$ for all $i \neq j$
    \item \textbf{Normalization:} $\|e_k\|_H = 1$ for all $k \in B$
    \item \textbf{Completeness:} The linear span of $\{e_k\}$ is dense in $H$ (closedness condition)
\end{enumerate}

Such a basis allows every element $f \in H$ to be represented as:
\[
f = \sum_{k \in B} \langle f, e_k \rangle e_k
\]
with convergence in the Hilbert space norm.
\end{definition}

\subsection{Important Properties in Inner Product Spaces}

\begin{theorem}[Cauchy-Schwarz Inequality]
For any elements $f, g$ in an inner product space, we have:
\[
|\langle f, g \rangle| \leq \|f\| \|g\|
\]
with equality if and only if $f$ and $g$ are linearly dependent.
\end{theorem}

\begin{theorem}[Pythagorean Theorem]
If $f \perp g$ (i.e., $\langle f, g \rangle = 0$), then:
\[
\|f + g\|^2 = \|f\|^2 + \|g\|^2
\]
\end{theorem}

\begin{theorem}[Parallelogram Law]
\[
2\|f\|^2 + 2\|g\|^2 = \|f + g\|^2 + \|f - g\|^2
\]
\end{theorem}

\section{Linear Operators}

\subsection{Bounded and Continuous Operators}

\begin{definition}[Linear Operator]
A map $A: F \to G$ between vector spaces is a linear operator if:
\[
A(\alpha_1 f_1 + \alpha_2 f_2) = \alpha_1 A(f_1) + \alpha_2 A(f_2)
\]
for all scalars $\alpha_1, \alpha_2$ and elements $f_1, f_2 \in F$.
\end{definition}

When $G = \mathbb{R}$ (or $\mathbb{C}$), a linear operator is called a functional.

\begin{definition}[Operator Norm]
The operator norm of a bounded linear operator $A : F \to G$ is:
\[
\|A\| := \sup_{f \in F, f \neq 0} \frac{\|A f\|_G}{\|f\|_F} = \sup_{\|f\|_F \leq 1} \|A f\|_G
\]

If $\|A\| < \infty$, we say $A$ is a bounded linear operator.
\end{definition}

\begin{theorem}[Equivalence of Boundedness and Continuity]
For a linear operator $A : F \to G$ between normed spaces, the following are equivalent:

\begin{enumerate}
    \item $A$ is a bounded operator
    \item $A$ is continuous on all of $F$
    \item $A$ is continuous at a single point in $F$
\end{enumerate}
\end{theorem}

\subsection{The Riesz Representation Theorem}

\begin{theorem}[Riesz Representation Theorem]
Let $H$ be a Hilbert space and let $f : H \to \mathbb{R}$ be a bounded linear functional. Then there exists a unique $y \in H$ such that:
\[
f(x) = \langle x, y \rangle_H \quad \text{for all } x \in H
\]
Moreover, $\|f\|_{H'} = \|y\|_H$, where $\|f\|_{H'}$ is the operator norm of $f$ on the dual space.
\end{theorem}

\begin{remark}
This theorem is fundamental: it says that every bounded linear functional on a Hilbert space can be represented as an inner product with some fixed vector. The mapping $J: H \to H'$ defined by $J(y)(x) = \langle x, y \rangle_H$ is an isometric isomorphism between $H$ and its dual space $H'$.
\end{remark}

\part{Kernels and Reproducing Kernel Hilbert Spaces}

\section{Kernels: Definition and Properties}

\subsection{Kernel Definition and Positive Definiteness}

\begin{definition}[Kernel]
Let $X$ be a non-empty set. A function $k : X \times X \to \mathbb{R}$ is a kernel if there exists a Hilbert space $H$ and a feature map $\phi : X \to H$ such that:
\[
k(x, x') = \langle \phi(x), \phi(x') \rangle_H \quad \text{for all } x, x' \in X
\]
\end{definition}

Equivalently, a kernel can be characterized through positive definiteness.

\begin{definition}[Positive Definite Kernel]
A function $k : X \times X \to \mathbb{R}$ is positive definite if for any finite set of points $\{x_1, \ldots, x_n\} \subseteq X$ and coefficients $\{c_1, \ldots, c_n\} \subseteq \mathbb{R}$:
\[
\sum_{i,j=1}^{n} c_i c_j k(x_i, x_j) \geq 0
\]
\end{definition}

The key insight is that kernels and positive definite functions are equivalent concepts.

\subsection{Constructing Kernels}

\begin{proposition}[Operations Preserving Kernels]
The following operations preserve the kernel property:

\begin{enumerate}
    \item \textbf{Positive scaling:} If $k$ is a kernel and $\alpha > 0$, then $\alpha k$ is a kernel.
    
    \item \textbf{Sums of kernels:} If $k_1$ and $k_2$ are kernels on $X$, then $k_1 + k_2$ is a kernel on $X$.
    
    \item \textbf{Products of kernels:} If $k_1$ is a kernel on $X_1$ and $k_2$ is a kernel on $X_2$, then the product 
    \[
    k(x, x') := k_1(x_1, x_1') \cdot k_2(x_2, x_2')
    \]
    is a kernel on $X_1 \times X_2$.
    
    \item \textbf{Kernel composition:} If $A: X \to \tilde{X}$ is a map and $\tilde{k}$ is a kernel on $\tilde{X}$, then $k(x, x') := \tilde{k}(A(x), A(x'))$ is a kernel on $X$.
\end{enumerate}
\end{proposition}

\begin{remark}
Not all differences of kernels are kernels. Additionally, a single kernel can admit infinitely many different feature map representations.
\end{remark}

\subsection{Examples of Kernels}

\begin{definition}[Polynomial Kernel]
The polynomial kernel of degree $d$ is defined as:
\[
k(x, x') := (\langle x, x' \rangle + c)^d
\]
where $d$ is a positive integer and $c \geq 0$ is a constant. This kernel corresponds to the feature space of all monomials of degree up to $d$ in the original features.
\end{definition}

\begin{definition}[Gaussian (RBF) Kernel]
The Gaussian or Radial Basis Function (RBF) kernel is:
\[
k(x, x') := \exp\left( -\gamma \|x - x'\|^2 \right)
\]
where $\gamma > 0$ is the bandwidth parameter. This kernel is universal and characteristic for probability measures.
\end{definition}

\begin{definition}[Kernels from Power Series]
For $r \in (0, \infty]$ and coefficients $\{a_n\}_{n=0}^{\infty}$ with $a_n \geq 0$, if the radius of convergence allows, we can define:
\[
k(x, x') := \sum_{n=0}^{\infty} a_n \langle x, x' \rangle^n
\]
for $\langle x, x' \rangle \in (-r, r)$.
\end{definition}

\section{Reproducing Kernel Hilbert Spaces (RKHS)}

\subsection{Evaluation Functionals}

\begin{definition}[Evaluation Functional]
Given a space $H$ of functions $f: X \to \mathbb{R}$ and a fixed point $x_0 \in X$, the evaluation functional at $x_0$ is:
\[
E_{x_0} : H \to \mathbb{R}, \quad E_{x_0}(f) := f(x_0)
\]

This is a linear functional, but it is not always bounded with respect to every norm on $H$.
\end{definition}

\subsection{RKHS Definition}

\begin{definition}[Reproducing Kernel Hilbert Space]
A Hilbert space $H$ of functions $f: X \to \mathbb{R}$ is a Reproducing Kernel Hilbert Space (RKHS) if for every $x \in X$, the evaluation functional $E_x : H \to \mathbb{R}$ defined by $E_x(f) = f(x)$ is bounded.

Equivalently, there exists a family $\{k_x\}_{x \in X} \subseteq H$ such that:
\[
f(x) = \langle f, k_x \rangle_H \quad \text{for all } x \in X, \, f \in H
\]

where $k_x(y) = k(y, x)$ and $k: X \times X \to \mathbb{R}$ is the reproducing kernel.
\end{definition}

This definition encodes the \textbf{reproducing property}: function values can be recovered via inner products with kernel sections.

\subsection{Moore-Aronszajn Theorem}

\begin{theorem}[Moore-Aronszajn]
There is a one-to-one correspondence between positive definite kernels on $X \times X$ and Reproducing Kernel Hilbert Spaces on $X$. 

Specifically, for every positive definite kernel $k$ on $X \times X$, there exists a unique RKHS $H_k$ with reproducing kernel $k$. Conversely, every RKHS has a unique reproducing kernel.

The RKHS $H_k$ is the completion of the pre-Hilbert space spanned by $\{k_x\}_{x \in X}$ with inner product:
\[
\left\langle \sum_i \alpha_i k_{x_i}, \sum_j \beta_j k_{x_j} \right\rangle_{H_k} := \sum_{i,j} \alpha_i \beta_j k(x_i, x_j)
\]
\end{theorem}

Key consequences:
\begin{itemize}
    \item Every positive definite kernel defines a unique RKHS
    \item The feature map $\phi(x) = k(\cdot, x)$ embeds $X$ into $H_k$
    \item The kernel value is $k(x, x') = \langle \phi(x), \phi(x') \rangle_{H_k}$
\end{itemize}

\subsection{Properties of RKHS and Smoothness}

\begin{proposition}[Smoothness from Small RKHS Norm]
If $f \in H_k$ with small $\|f\|_{H_k}$, then $f$ is smooth. Specifically:
\[
|f(x) - f(x')| = |\langle f, k_x - k_{x'} \rangle| \leq \|f\|_{H_k} \|k_x - k_{x'}\|_{H_k}
\]

For smooth kernels, $\|k_x - k_{x'}\|_{H_k}$ is small when $x$ is close to $x'$.
\end{proposition}

\section{Spectral Properties and Function Representations}

\subsection{Integral Linear Operators and Mercer's Theorem}

\begin{definition}[Integral Operator]
Given a kernel $k$ and a measure $\nu$ on $X$, the integral operator is:
\[
(T_k f)(x) := \int_X k(x, y) f(y) \, d\nu(y)
\]
This is a linear operator mapping appropriate function spaces.
\end{definition}

\begin{theorem}[Mercer's Theorem]
Let $k$ be a continuous kernel on a compact metric space $X$, and let $\nu$ be a finite Borel measure with $\text{supp}(\nu) = X$. Then $k$ admits an eigenfunction expansion:
\[
k(x, y) = \sum_{j=1}^{\infty} \lambda_j e_j(x) e_j(y)
\]

where $\lambda_1 \geq \lambda_2 \geq \ldots \geq 0$ are eigenvalues of the integral operator $T_k$ and $\{e_j\}$ are the corresponding orthonormal eigenfunctions. The series converges absolutely and uniformly on $X \times X$.
\end{theorem}

\subsection{Spectral Theorem and Decomposition}

\begin{theorem}[Spectral Decomposition]
Let $H$ be a Hilbert space and $T: H \to H$ a compact, self-adjoint operator. Then there exists an at-most countable orthonormal set $\{e_j\}_{j \in J}$ and eigenvalues $\lambda_1 \geq \lambda_2 \geq \ldots$ converging to zero, such that:
\[
f = \sum_{j \in J} \langle f, e_j \rangle e_j \quad \text{and} \quad T f = \sum_{j \in J} \lambda_j \langle f, e_j \rangle e_j
\]
for all $f \in H$.
\end{theorem}

This allows us to represent any function in $H_k$ and understand the decay of the spectrum.

\part{Applications of Kernels}

\section{Kernel Methods in Machine Learning}

\subsection{Representer Theorem}

\begin{theorem}[Representer Theorem]
Consider a regularized empirical risk minimizer:
\[
\min_{f \in H_k} \frac{1}{n} \sum_{i=1}^{n} L(y_i, f(x_i)) + \lambda \|f\|_{H_k}^2
\]

where $L$ is any loss function and $\lambda > 0$. The optimal solution has the form:
\[
f^*(x) = \sum_{i=1}^{n} \alpha_i^* k(x, x_i)
\]

That is, the optimal function is expressible as a linear combination of kernel evaluations at the training points.
\end{theorem}

This is why kernel methods are computationally practical: we only need to optimize over $n$ coefficients rather than solving in infinite-dimensional spaces.

\subsection{Kernel Ridge Regression}

\begin{definition}[Kernel Ridge Regression]
Kernel Ridge Regression solves:
\[
\min_{f \in H_k} \frac{1}{n} \sum_{i=1}^{n} (y_i - f(x_i))^2 + \lambda \|f\|_{H_k}^2
\]

By the representer theorem, the solution is:
\[
f^*(x) = \sum_{i=1}^{n} \alpha_i^* k(x, x_i)
\]

where $\boldsymbol{\alpha}^* = (K + n\lambda I)^{-1}\mathbf{y}$ and $K$ is the kernel matrix with entries $K_{ij} = k(x_i, x_j)$.
\end{definition}

\section{Kernel Mean Embeddings}

\subsection{Definition and Properties}

\begin{definition}[Kernel Mean Embedding]
For a probability measure $P$ on $\mathcal{X}$ and kernel $k$, the kernel mean embedding is:
\[
\mu_P := \mathbb{E}_{X \sim P}[k(\cdot, X)] = \int k(\cdot, x) \, dP(x) \in H_k
\]

For an empirical measure $P_n = \frac{1}{n}\sum_{i=1}^{n} \delta_{x_i}$, the empirical mean embedding is:
\[
\hat{\mu}_{P_n} := \frac{1}{n} \sum_{i=1}^{n} k(\cdot, x_i)
\]
\end{definition}

Mean embeddings map probability distributions to function space, preserving distributional information.

\begin{proposition}[Injectivity for Characteristic Kernels]
A kernel $k$ is called characteristic for probability measures on $\mathcal{X}$ if the embedding $P \mapsto \mu_P$ is injective. That is:
\[
\mu_P = \mu_Q \implies P = Q
\]

The Gaussian kernel and many translation-invariant kernels with full support are characteristic.
\end{proposition}

\section{Maximum Mean Discrepancy (MMD)}

\subsection{Definition}

\begin{definition}[Maximum Mean Discrepancy]
The MMD between two probability measures $P$ and $Q$ with a kernel $k$ is:
\[
\text{MMD}_k(P, Q) := \|\mu_P - \mu_Q\|_{H_k}
\]

Equivalently:
\[
\text{MMD}_k^2(P, Q) = \mathbb{E}_{x,x' \sim P}[k(x, x')] - 2\mathbb{E}_{x \sim P, y \sim Q}[k(x, y)] + \mathbb{E}_{y,y' \sim Q}[k(y, y')]
\]
\end{definition}

MMD provides a metric on the space of probability measures via an RKHS.

\subsection{Witness Function and Interpretation}

The function that witnesses the maximum discrepancy—the one that achieves the supremum in the definition—is the normalized difference of mean embeddings:
\[
f^*(x) = \frac{\mu_P(x) - \mu_Q(x)}{\|\mu_P - \mu_Q\|_{H_k}}
\]

This function shows where the distributions differ most in the feature space.

\subsection{Two-Sample Testing with MMD}

\begin{proposition}[Hypothesis Testing]
Under the null hypothesis $H_0: P = Q$, the statistic MMD$_k^2(P_n, Q_m)$ converges to 0. The asymptotic distribution can be derived for test construction.

For large samples and characteristic kernels, MMD provides a consistent test for distributional differences.
\end{proposition}

\section{Hilbert-Schmidt Independence Criterion (HSIC)}

\subsection{Constrained Covariance}

\begin{definition}[Constrained Covariance]
For two probability measures $P$ on spaces with kernels $k$ and $\ell$, the constrained covariance is:
\[
\text{COCO}(P) := \sup_{\|f\|_{H_k}=1, \|g\|_{H_\ell}=1} \text{Cov}[f(X), g(Y)]
\]

where $\text{Cov}[f(X), g(Y)] = \mathbb{E}[f(X)g(Y)] - \mathbb{E}[f(X)]\mathbb{E}[g(Y)]$.
\end{definition}

\subsection{HSIC Definition}

\begin{definition}[Hilbert-Schmidt Independence Criterion]
The HSIC between random variables $X$ (with kernel $k$) and $Y$ (with kernel $\ell$) is:
\[
\text{HSIC}(P_{XY}; H_k, H_\ell) := \|\mathcal{C}_{XY}\|_{HS}^2
\]

where $\mathcal{C}_{XY}$ is the cross-covariance operator in the tensor product Hilbert space.
\end{definition}

HSIC measures statistical dependence and is zero if and only if $X \perp Y$.

\subsection{Empirical HSIC}

For samples $\{(x_i, y_i)\}_{i=1}^{n}$, the empirical HSIC can be computed efficiently as:
\[
\widehat{\text{HSIC}} = \frac{1}{n^2} \text{tr}(KHLH)
\]

where $K$ and $\ell$ are the kernel matrices and $H = I - \frac{1}{n}\mathbf{1}\mathbf{1}^T$ is the centering matrix.

\part{Advanced Topics}

\section{Kernel Stein Discrepancy}

\subsection{Stein Operators}

\begin{definition}[Langevin Stein Operator]
For a probability distribution $p$ with score function $\nabla \log p(x)$, the Langevin Stein operator is:
\[
(\mathcal{A}_p f)(x) := \nabla \log p(x) \cdot f(x) + \nabla f(x)
\]

This operator has the key property that:
\[
\mathbb{E}_p[\mathcal{A}_p f] = 0
\]
for all appropriate test functions $f$.
\end{definition}

\subsection{Definition of KSD}

\begin{definition}[Kernel Stein Discrepancy]
The Kernel Stein Discrepancy (KSD) measures the maximum violation of the Stein equation:
\[
\text{KSD}(Q, p; H_k) := \sup_{\|f\|_{H_k} \leq 1} \left| \mathbb{E}_{X \sim Q}[(\mathcal{A}_p f)(X)] \right|
\]

KSD is zero if and only if $Q = p$, making it a proper divergence for distribution testing without density ratio estimation.
\end{definition}

KSD is particularly useful when we have an unnormalized density or samples from a distribution but cannot compute the likelihood.

\section{Optimization and Convex Analysis}

\subsection{Lagrangian and Duality}

\begin{definition}[Lagrangian]
For a constrained optimization problem:
\[
\min_x f_0(x) \quad \text{subject to } f_i(x) \leq 0, \quad i=1,\ldots,m, \quad h_j(x) = 0, \quad j=1,\ldots,p
\]

the Lagrangian is:
\[
L(x, \lambda, \nu) := f_0(x) + \sum_{i=1}^{m} \lambda_i f_i(x) + \sum_{j=1}^{p} \nu_j h_j(x)
\]

where $\lambda_i \geq 0$ are inequality constraint multipliers and $\nu_j$ are equality constraint multipliers.
\end{definition}

The dual problem maximizes:
\[
g(\lambda, \nu) := \inf_x L(x, \lambda, \nu)
\]
subject to $\lambda \geq 0$.

\subsection{Strong Duality and KKT Conditions}

\begin{theorem}[Strong Duality via Slater's Condition]
If the primal problem is convex and Slater's condition holds (there exists a feasible point with strict inequality for inequality constraints), then strong duality holds: the optimal value of the primal equals the optimal value of the dual.
\end{theorem}

\begin{theorem}[KKT Conditions]
A point $x^*$ with multipliers $(\lambda^*, \nu^*)$ satisfies the KKT conditions if:

\begin{enumerate}
    \item \textbf{Stationarity:} $0 \in \partial f_0(x^*) + \sum_i \lambda_i^* \partial f_i(x^*) + \sum_j \nu_j^* \partial h_j(x^*)$
    
    \item \textbf{Complementary slackness:} $\lambda_i^* f_i(x^*) = 0$ for all $i$
    
    \item \textbf{Primal feasibility:} $f_i(x^*) \leq 0$ and $h_j(x^*) = 0$
    
    \item \textbf{Dual feasibility:} $\lambda_i^* \geq 0$ for all $i$
\end{enumerate}

For convex problems with strong duality, the KKT conditions characterize optimal solutions.
\end{theorem}

\textbf{Complementary slackness} is key: if $\lambda_i^* > 0$, then $f_i(x^*) = 0$ (the constraint is active).
